Продажа и возврат ​

js
it('Продажа и возврат билета', () => {
  cy.login('kassir1', 'password');
  cy.selectTrip('Ростов → Казань', '2025-04-10');
  cy.selectSeat(12);
  cy.sellTicket('наличные');
  cy.returnTicket('Не приехал');
  cy.contains('Возврат успешно выполнен');
});
